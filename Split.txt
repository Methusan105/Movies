read -r -p "Enter MP4 path: " FILE; FILE="${FILE//[\'\"]}"; ffmpeg -i "$FILE" -c copy -map 0 -segment_time $(ffprobe -v error -show_entries format=duration -of default=noprint_wrappers=1:nokey=1 "$FILE" | awk '{print int(1850 * $1 / ('$(stat -c%s "$FILE")' / 1048576))}') -f segment -segment_start_number 1 -reset_timestamps 1 "${FILE%.*}_%02d.${FILE##*.}"
